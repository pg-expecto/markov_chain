# Ежедневная очистка журналов и обновление статистик
# ============================================================
# Очистка transition_log (в 01:15)
15 1 * * * psql -d performance_monitoring_db -U performance_monitoring_user -c "SELECT mchain_clean_transition_log();"

# ============================================================
# Очистка архивных данных (раз в неделю)
# ============================================================
# Очистка apply_forgetting_log (ежедневно в 02:00)
0 2 * * * psql -d performance_monitoring_db -U performance_monitoring_user -c "SELECT mchain_clean_apply_forgetting_log();"

# ============================================================
# Анализ качества прогноза
# ============================================================

# Обновление critical_states еженедельно (суббота в 03:00)
0 3 * * 6 psql -d performance_monitoring_db -U performance_monitoring_user -c "SELECT refresh_critical_states();" >/postgres/pg_expecto/sh/refresh_critical_states.log 2>&1

# Обновление исходов каждые 5 минут 
*/5 * * * * psql -d performance_monitoring_db -U performance_monitoring_user -c "SELECT update_prediction_outcomes();"

# Расчёт суточных метрик в 02:00
0 2 * * * psql -d performance_monitoring_db -U performance_monitoring_user -c "SELECT calculate_daily_quality_metrics(CURRENT_DATE - 1);"

# Расчет метрик профиля нагрузки на основе цепи Маркова 
# Еженедельная адаптивная настройка min_freq_for_stability (воскресенье в 01:00)
0 1 * * 0 psql -d performance_monitoring_db -U performance_monitoring_user -c "SELECT refresh_stability_threshold();"

# Расчет метрик профиля нагрузки на основе цепи Маркова 
# Оперативный профиль (каждые 5 минут)
*/5 * * * * psql -d performance_monitoring_db -U performance_monitoring_user -c "SELECT save_operational_profile();" > /postgres/pg_expecto/sh/save_operational_profile.log 2>&1

# Суточный профиль (каждый час)
0 * * * * psql -d performance_monitoring_db -U performance_monitoring_user -c "SELECT save_daily_profile();" > /postgres/pg_expecto/sh/save_daily_profile.log 2>&1

# Недельный профиль (раз в день, в 01:00)
0 1 * * * psql -d performance_monitoring_db -U performance_monitoring_user -c "SELECT save_weekly_profile();" > /postgres/pg_expecto/sh/save_weekly_profile.log  2>&1

# Очистка старых профилей
# Оперативные профили можно хранить 7 дней, суточные – 30 дней, недельные – 90 дней.
# Удаление оперативных профилей старше 7 дней (каждый день в 02:00)
0 2 * * * psql -d performance_monitoring_db -U performance_monitoring_user -c "DELETE FROM profile_aggregated WHERE profile_type='operational' AND ts < now() - interval '7 days';" > /postgres/pg_expecto/sh/operational.log 2>&1

# Удаление суточных профилей старше 30 дней
0 3 * * * psql -d performance_monitoring_db -U performance_monitoring_user -c "DELETE FROM profile_aggregated WHERE profile_type='daily' AND ts < now() - interval '30 days';" > /postgres/pg_expecto/sh/daily.log 2>&1

# Удаление недельных профилей старше 90 дней
0 4 * * * psql -d performance_monitoring_db -U performance_monitoring_user -c "DELETE FROM profile_aggregated WHERE profile_type='weekly' AND ts < now() - interval '90 days';" > /postgres/pg_expecto/sh/weekly.log 2>&1

# Обновление performance_history каждую минуту
*/1 * * * * psql -d performance_monitoring_db -U performance_monitoring_user -c "SELECT refresh_performance_history();" > /postgres/pg_expecto/sh/refresh_performance_history.log 2>&1

# Ежедневная очистка (в 03:00)
0 3 * * * psql -d performance_monitoring_db -U performance_monitoring_user -c "SELECT clean_profile_comparison_log();"

# Ежедневно в 02:30 добавляем профили для инцидентов за предыдущий день
# Для первоначального заполнения за более длительный период (например, за 30 дней) можно выполнить вручную:
#SELECT collect_pre_incident_profiles(now() - interval '30 days', now(), 60);
30 2 * * * psql -d performance_monitoring_db -U performance_monitoring_user -c "SELECT collect_pre_incident_profiles(now() - interval '1 day', now(), 60);" > /postgres/pg_expecto/sh/collect_pre_incident.log 2>&1

# Очистка старых данных индикатора изменения профиля (каждый день в 2:00)
0 2 * * * psql -d performance_monitoring_db -U performance_monitoring_user -c "SELECT clean_old_profile_change_indicator(30);" > /postgres/pg_expecto/sh/clean_old_profile_change_indicator.log 2>&1
